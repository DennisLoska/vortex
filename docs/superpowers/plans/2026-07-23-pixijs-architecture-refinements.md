```ts
import type { Manifest, ManifestAsset } from "../manifest";
// eslint-disable-next-line @typescript-eslint/ban-ts-comment
// @ts-ignore - dynamically generated by AssetPack
import _manifest from "../manifest.json";

const manifest = _manifest as Manifest;

const IMAGE_EXTS = [".png", ".jpg", ".jpeg", ".webp", ".svg"];
const VIDEO_EXTS = [".mp4", ".webm", ".m4v", ".ogv", ".mov"];
const GIF_EXT = ".gif";

export type PoolEntry = {
  key: string;
  type: "image" | "video" | "gif";
};

export function getProjectNames(): string[] {
  const names = new Set<string>();
  for (const bundle of manifest.bundles) {
    for (const asset of bundle.assets) {
      for (const src of asset.src) {
        const first = src.split("/")[0];
        if (first) names.add(first);
      }
    }
  }
  return [...names];
}

export function getProjectAssets(project: string): PoolEntry[] {
  const pool: PoolEntry[] = [];
  const seen = new Set<string>();
  const defaultBundle = manifest.bundles.find((b) => b.name === "default");
  const assets = defaultBundle?.assets ?? [];

  for (const asset of assets) {
    const firstSrc = asset.src[0];
    const lower = firstSrc.toLowerCase();
    const key = asset.alias[0] ?? firstSrc;

    if (seen.has(key)) continue;
    if (!key.startsWith(project)) continue;
    if (key.startsWith(`${project}/backgrounds/`)) continue;

    if (IMAGE_EXTS.some((ext) => lower.endsWith(ext))) {
      seen.add(key);
      pool.push({ key, type: "image" });
    } else if (VIDEO_EXTS.some((ext) => lower.endsWith(ext))) {
      seen.add(key);
      pool.push({ key, type: "video" });
    } else if (lower.endsWith(GIF_EXT)) {
      seen.add(key);
      pool.push({ key, type: "gif" });
    }
  }
  return pool;
}

export function getProjectBackgrounds(project: string): {
  videoAliases: string[];
  imageAliases: string[];
} {
  const videoAliases: string[] = [];
  const imageAliases: string[] = [];

  for (const bundle of manifest.bundles) {
    for (const asset of bundle.assets) {
      const firstSrc = asset.src[0];
      if (!firstSrc.startsWith(`${project}/backgrounds/`)) continue;

      const alias = asset.alias[0] ?? firstSrc;
      const ext = firstSrc.split(".").pop()?.toLowerCase();

      if (ext === "mp4" || ext === "webm" || ext === "ogg") {
        videoAliases.push(alias);
      } else if (ext === "png" || ext === "jpg" || ext === "jpeg" || ext === "webp") {
        imageAliases.push(alias);
      }
    }
  }
  return { videoAliases, imageAliases };
}

export { manifest };
```

- [ ] **Step 2: Refactor `setMultipleBackgrounds`**

Replace the method. Instead of parsing manifest and calling `loadAllInBackground`, use `Assets.loadBundle`:

```ts
public async setMultipleBackgrounds(projectName: string) {
  this.loadGen++;
  const gen = this.loadGen;
  this.textures = [];
  this.videos = [];

  const { videoAliases, imageAliases } = getProjectBackgrounds(projectName);

  const allAliases = [...videoAliases, ...imageAliases];

  if (allAliases.length === 0) return;

  // Define a unique bundle name per project+gen so old loads don't collide
  const bundleName = `${projectName}-bg-${gen}`;

  // Register as a temp bundle for background loading
  Assets.addBundle(bundleName, allAliases);

  // Load with concurrency (Assets loads with its own internal queue)
  await Assets.loadBundle(bundleName, (progress) => {
    // track progress if needed; minimal usage here
  });

  // Guard against stale load generation
  if (gen !== this.loadGen) {
    Assets.unloadBundle(bundleName);
    return;
  }

  // Collect textures and video elements from loaded assets
  for (const alias of allAliases) {
    const tex = Assets.get(alias);
    if (!tex) continue;
    this.textures.push(tex);

    // Track video elements for playback control
    const source = (tex as Texture).source;
    if (source instanceof VideoSource) {
      const videoEl = source.resource as HTMLVideoElement;
      if (videoEl) {
        videoEl.loop = true;
        videoEl.muted = true;
        videoEl.playsInline = true;
        this.videos.push(videoEl);
        void videoEl.play();
      }
    }
  }

  if (this.textures.length === 0) return;

  this.currentIdx = 0;

  this.activeSprite = new Sprite({
    texture: this.textures[0],
    anchor: 0.5,
    alpha: 0,
  });
  this.addChild(this.activeSprite);
  this.stretchToFill(this.activeSprite);

  this.needsFadeIn = true;
  this.fadeInAlpha = 0;
  this.autoTimer = 0;
  this.nextAutoDelay = 999;
}
```

- [ ] **Step 1: Refactor `src/engine/resize/ResizePlugin.ts`**

Replace the entire file content:

```ts
import { ExtensionType, ResizePlugin } from "pixi.js";
import type { Application, ApplicationOptions, ExtensionMetadata } from "pixi.js";

import { resize } from "./resize";

declare module "pixi.js" {
  interface Application {
    resizeOptions: {
      minWidth: number;
      minHeight: number;
      letterbox: boolean;
    };
  }
}

export class CreationResizePlugin {
  public static extension: ExtensionMetadata = ExtensionType.Application;

  public static init(options: ApplicationOptions): void {
    // Let stock ResizePlugin set up resizeTo, queueResize, cancelResize,
    // event listeners, and the basic resize function
    ResizePlugin.init.call(this, options);

    const app = this as unknown as Application;

    // Add custom resize options
    const resizeOpts = (options as any).resizeOptions ?? {};
    app.resizeOptions = {
      minWidth: resizeOpts.minWidth ?? 768,
      minHeight: resizeOpts.minHeight ?? 1024,
      letterbox: resizeOpts.letterbox ?? true,
    };

    // Override resize with custom min-size + letterbox logic
    app.resize = (): void => {
      const self = this as any;
      if (!self._resizeTo) return;

      self._cancelResize();

      let canvasWidth: number;
      let canvasHeight: number;

      if (self._resizeTo === globalThis.window) {
        canvasWidth = globalThis.innerWidth;
        canvasHeight = globalThis.innerHeight;
      } else {
        const { clientWidth, clientHeight } = self._resizeTo as HTMLElement;
        canvasWidth = clientWidth;
        canvasHeight = clientHeight;
      }

      const { width, height } = resize(
        canvasWidth,
        canvasHeight,
        app.resizeOptions.minWidth,
        app.resizeOptions.minHeight,
        app.resizeOptions.letterbox,
      );

      app.renderer.canvas.style.width = `${canvasWidth}px`;
      app.renderer.canvas.style.height = `${canvasHeight}px`;
      window.scrollTo(0, 0);
      app.renderer.resize(width, height);
    };
  }

  public static destroy(): void {
    ResizePlugin.destroy.call(this);
  }
}
```
